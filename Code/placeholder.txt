#code for the project:belt drive sander machine.

  
// Pin Definitions
const int irSensorPin = 2;        // IR sensor connected to digital pin 2
const int motorPin1 = 8;          // Motor input pin 1
const int motorPin2 = 9;          // Motor input pin 2

void setup() {
  pinMode(irSensorPin, INPUT);
  pinMode(motorPin1, OUTPUT);
  pinMode(motorPin2, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  int irValue = digitalRead(irSensorPin);
  
  if (irValue == LOW) {  // Object detected (many IR sensors give LOW when object is detected)
    Serial.println("Object Detected: Motor ON (Clockwise)");
    digitalWrite(motorPin1, HIGH);
    digitalWrite(motorPin2, LOW);   // Clockwise rotation
  } else {
    Serial.println("No Object: Motor OFF");
    digitalWrite(motorPin1, LOW);
    digitalWrite(motorPin2, LOW);   // Stop motor
  }

  delay(100);
}
